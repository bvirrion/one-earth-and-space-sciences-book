\section*{Chapter \ref{ch:g1:watching-the-weather} --- Watching the Weather}

\begin{solution}{exo:g1:watching-the-weather:1}
Sunny: \omwx{sun}\quad rainy: \omwx{rain}\quad snowy: \omwx{snow}\quad
cloudy: \omwx{cloud}. Any clear drawing of a sun, a cloud with drops, a
cloud with snowflakes and a cloud alone is right.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:2}
The park: sunny and warm. The street: snowy and cold. The hill: windy (and
some clouds). The school gate: foggy.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:3}
Tuesday morning: cloudy. Friday afternoon: showers. Sunday afternoon:
rainy.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:4}
Wednesday: the sun is drawn in the morning and in the afternoon.
(On Thursday the afternoon has sun and clouds.)
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:5}
The sun hat goes with hot and sunny \omterm{def:g1:watching-the-weather:weather}{weather}; boots and a raincoat with
rainy \omterm{def:g1:watching-the-weather:weather}{weather}; a scarf and gloves with cold and snowy \omterm{def:g1:watching-the-weather:weather}{weather}.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:6}
Windy \omterm{def:g1:watching-the-weather:weather}{weather}: the wind pushes the flag.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:7}
\begin{enumerate}
  \item False: the \omterm{def:g1:watching-the-weather:weather}{weather} can change during the day, for example a grey
        morning and a sunny afternoon.
  \item True.
  \item True: the Sun is so bright that it can hurt the eye for ever,
        even in a moment.
\end{enumerate}
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:8}
Afternoons with rain or showers: Monday (rain), Friday (showers) and Sunday (rain), so 3.
Cold days: Monday, Tuesday and Sunday, so 3. The other 4 days were warm, so
the week was mostly warm.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:9}
Morning: \omwx{cloud} (grey sky); lunchtime: \omwx{rain} (rain);
evening: \omwx{suncloud} (blue sky with a few clouds).
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:10}
A flag points to where the wind is going. If it points to the left, the
wind is going to the left, so it comes from the right.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:11}
No leaves: in winter. Blossom: in spring. Winter has the snow symbol: in
winter it is cold, and in many places snow falls only in that season.
\end{solution}
