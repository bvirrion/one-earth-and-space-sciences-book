\section*{Hoofdstuk \ref{ch:g2:drawing-to-map} --- Van tekening tot kaart}

\begin{solution}{exo:g2:drawing-to-map:1}
Zoals je het van bovenaf ziet.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:2}
De blauwe lijnen bij de westmuur zijn de ramen. In de noordoostelijke hoek
staat de kast.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:3}
Een titel, een \omterm{def:g2:drawing-to-map:legend}{legenda}, een \omterm{def:g2:drawing-to-map:scale}{schaal} en een \omterm{def:g2:drawing-to-map:directions}{windroos} (of een pijl die naar het noorden wijst).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:4}
Het noorden ligt bovenaan; het westen ligt links.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:5}
Eén hokje staat voor 1 meter: de tafel is 2 meter lang en 1 meter breed.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:6}
Het bruine vierkantje is de boerderij; de groene vlek is het bos.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:7}
Het bos ligt ten zuiden van de \omterm{def:g1:landscapes-around-us:water}{rivier} en ten oosten van de boerderij.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:8}
Eén hokje staat voor 1 kilometer: 10 kilometer van west naar oost en
8 kilometer van zuid naar noord.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:9}
Drie rijen van drie tafels: 9 tafels. De deur zit in het zuidelijke deel
van de oostmuur, dus de tafel die het dichtst bij de deur staat, staat in
het zuiden (in de zuidoostelijke hoek van het lokaal).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:10}
Het bed is 2 hokjes lang. De kamer is 4 hokjes bij 3 hokjes: 3 rijen van
4 hokjes, dat is $4 + 4 + 4 = 12$ hokjes.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:11}
``Links'' hangt ervan af welke kant Lena op kijkt: als ze zich omdraait,
ligt de \omterm{def:g1:landscapes-around-us:water}{rivier} rechts van haar. Voor iedereen: ``de \omterm{def:g1:landscapes-around-us:water}{rivier} ligt ten noorden
van de boerderij''.
\end{solution}
