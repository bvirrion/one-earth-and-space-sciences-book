\section*{Bab \ref{ch:g2:drawing-to-map} --- Dari Gambar ke Peta}

\begin{solution}{exo:g2:drawing-to-map:1}
Seperti dilihat dari atas.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:2}
Garis-garis biru di dinding barat adalah jendela. Di sudut timur laut ada
lemari.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:3}
Judul, \omterm{def:g2:drawing-to-map:legend}{legenda}, \omterm{def:g2:drawing-to-map:scale}{skala}, dan \omterm{def:g2:drawing-to-map:directions}{mawar angin} (atau panah yang menunjuk ke utara).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:4}
Utara ada di atas; barat ada di kiri.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:5}
Satu kotak mewakili 1 meter: panjang meja itu 2 meter dan lebarnya 1 meter.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:6}
Persegi cokelat adalah pertanian; bentuk hijau adalah hutan.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:7}
Hutan itu berada di sebelah selatan \omterm{def:g1:landscapes-around-us:water}{sungai}, dan di sebelah timur pertanian.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:8}
Satu kotak mewakili 1 kilometer: 10 kilometer dari barat ke timur dan 8
kilometer dari selatan ke utara.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:9}
Tiga baris yang masing-masing berisi tiga meja: 9 meja. Pintunya ada di
bagian selatan dinding timur, jadi meja yang paling dekat dengan pintu
berada di selatan (sudut tenggara ruangan).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:10}
Tempat tidur itu panjangnya 2 kotak. Kamarnya berukuran 4 kotak kali 3
kotak: 3 baris yang masing-masing berisi 4 kotak, yaitu $4 + 4 + 4 = 12$ kotak.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:11}
``Kiri'' bergantung pada arah Lena menghadap: jika ia berbalik, \omterm{def:g1:landscapes-around-us:water}{sungai} itu
ada di sebelah kanannya. Untuk semua orang: ``\omterm{def:g1:landscapes-around-us:water}{sungai} itu berada di sebelah
utara pertanian''.
\end{solution}
