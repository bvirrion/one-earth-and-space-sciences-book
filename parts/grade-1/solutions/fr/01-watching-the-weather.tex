\section*{Chapitre \ref{ch:g1:watching-the-weather} --- Observer le temps qu'il fait}

\begin{solution}{exo:g1:watching-the-weather:1}
Soleil~: \omwx{sun}\quad pluie~: \omwx{rain}\quad neige~: \omwx{snow}\quad
nuages~: \omwx{cloud}. Tout dessin clair d'un soleil, d'un nuage avec des
gouttes, d'un nuage avec des flocons et d'un nuage seul est juste.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:2}
Le parc~: du soleil, et il fait chaud. La rue~: de la neige, et il fait
froid. La colline~: du vent (et quelques nuages). L'entrée de l'école~: du
brouillard.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:3}
Mardi matin~: nuageux. Vendredi après-midi~: des averses. Dimanche
après-midi~: de la pluie.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:4}
Mercredi~: le soleil est dessiné le matin et l'après-midi.
(Jeudi après-midi, il y a du soleil et des nuages.)
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:5}
Le chapeau de soleil va avec un temps chaud et ensoleillé~; les bottes et
l'imperméable avec un temps pluvieux~; l'écharpe et les gants avec un temps
froid et neigeux.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:6}
Il y a du vent~: le vent pousse le drapeau.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:7}
\begin{enumerate}
  \item Faux~: le temps peut changer pendant la journée, par exemple un
        matin gris et un après-midi ensoleillé.
  \item Vrai.
  \item Vrai~: le Soleil est si lumineux qu'il peut abîmer l'œil pour
        toujours, même en un instant.
\end{enumerate}
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:8}
Après-midi avec de la pluie ou des averses~: lundi (pluie), vendredi (averses) et dimanche
(pluie), donc 3. Jours froids~: lundi, mardi et dimanche, donc 3. Les
4 autres jours ont été chauds, donc la semaine a été plutôt chaude.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:9}
Le matin~: \omwx{cloud} (ciel gris)~; à midi~: \omwx{rain} (pluie)~;
le soir~: \omwx{suncloud} (ciel bleu avec quelques nuages).
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:10}
Un drapeau montre où va le vent. S'il montre la gauche, le vent va vers la
gauche, donc il vient de la droite.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:11}
Pas de feuilles~: en hiver. En fleurs~: au printemps. L'hiver a le symbole
de la neige~: en hiver il fait froid, et dans beaucoup d'endroits, la neige
ne tombe qu'en cette saison.
\end{solution}
