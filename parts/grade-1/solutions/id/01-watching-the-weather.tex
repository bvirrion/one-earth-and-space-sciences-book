\section*{Bab \ref{ch:g1:watching-the-weather} --- Mengamati Cuaca}

\begin{solution}{exo:g1:watching-the-weather:1}
Cerah: \omwx{sun}\quad hujan: \omwx{rain}\quad bersalju: \omwx{snow}\quad
berawan: \omwx{cloud}. Setiap gambar yang jelas berupa matahari, awan
dengan tetes air, awan dengan butir salju, dan awan saja sudah benar.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:2}
Taman: cerah dan hangat. Jalan: bersalju dan dingin. Bukit: berangin (dan
ada beberapa awan). Gerbang sekolah: berkabut.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:3}
Selasa pagi: berawan. Jumat siang: hujan sebentar-sebentar. Minggu siang:
hujan.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:4}
Rabu: matahari digambar pada pagi hari dan juga pada siang hari.
(Pada hari Kamis, siangnya cerah berawan.)
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:5}
Topi pantai cocok dengan \omterm{def:g1:watching-the-weather:weather}{cuaca} panas dan cerah; sepatu bot dan jas hujan
dengan \omterm{def:g1:watching-the-weather:weather}{cuaca} hujan; syal dan sarung tangan dengan \omterm{def:g1:watching-the-weather:weather}{cuaca} dingin dan bersalju.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:6}
\omterm{def:g1:watching-the-weather:weather}{Cuaca} berangin: angin mendorong benderanya.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:7}
\begin{enumerate}
  \item Salah: \omterm{def:g1:watching-the-weather:weather}{cuaca} bisa berubah dalam satu hari, misalnya pagi yang
        kelabu lalu siang yang cerah.
  \item Benar.
  \item Benar: Matahari begitu terang sehingga bisa merusak mata
        selamanya, bahkan hanya dalam sekejap.
\end{enumerate}
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:8}
Siang yang hujan atau hujan sebentar-sebentar: Senin (hujan), Jumat (hujan
sebentar-sebentar), dan Minggu (hujan), jadi 3. Hari yang dingin: Senin,
Selasa, dan Minggu, jadi 3. Keempat hari lainnya hangat, jadi minggu itu
lebih sering hangat.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:9}
Pagi: \omwx{cloud} (langit kelabu); waktu makan siang: \omwx{rain} (hujan);
malam: \omwx{suncloud} (langit biru dengan beberapa awan).
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:10}
Bendera menunjuk ke arah angin pergi. Jika bendera menunjuk ke kiri, angin
pergi ke kiri, jadi angin datang dari kanan.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:11}
Tidak berdaun: di musim dingin. Berbunga: di musim semi. Musim dingin
diberi simbol salju: di musim dingin udara dingin, dan di banyak tempat
salju hanya turun pada musim itu.
\end{solution}
