\section*{Capítulo \ref{ch:g1:watching-the-weather} --- Observando o tempo}

\begin{solution}{exo:g1:watching-the-weather:1}
Ensolarado: \omwx{sun}\quad chuvoso: \omwx{rain}\quad com neve: \omwx{snow}\quad
nublado: \omwx{cloud}. Qualquer desenho claro de um sol, de uma nuvem com
gotas, de uma nuvem com flocos de neve e de uma nuvem sozinha está certo.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:2}
O parque: ensolarado e quente. A rua: com neve e frio. O morro: com vento
(e algumas nuvens). O portão da escola: com neblina.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:3}
Manhã de terça-feira: nublado. Tarde de sexta-feira: pancadas de chuva.
Tarde de domingo: chuvoso.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:4}
Quarta-feira: o sol está desenhado de manhã e à tarde.
(Na quinta-feira, a tarde tem sol e nuvens.)
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:5}
O chapéu de sol combina com o tempo quente e ensolarado; as botas e a
capa de chuva, com o tempo chuvoso; o cachecol e as luvas, com o tempo
frio e com neve.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:6}
Está ventando: o vento empurra a bandeira.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:7}
\begin{enumerate}
  \item Falso: o \omterm{def:g1:watching-the-weather:weather}{tempo} pode mudar durante o dia; por exemplo, uma manhã
        cinzenta e uma tarde de sol.
  \item Verdadeiro.
  \item Verdadeiro: o Sol é tão brilhante que pode machucar o olho para
        sempre, mesmo num instante.
\end{enumerate}
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:8}
Tardes com chuva ou pancadas de chuva: segunda (chuva), sexta-feira (pancadas) e domingo
(chuva), ou seja, 3. Dias frios: segunda, terça e domingo, ou seja, 3. Os
outros 4 dias foram quentes, então a semana foi mais quente.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:9}
Manhã: \omwx{cloud} (céu cinzento); hora do almoço: \omwx{rain} (chuva);
noite: \omwx{suncloud} (céu azul com algumas nuvens).
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:10}
Uma bandeira aponta para onde o vento vai. Se ela aponta para a esquerda,
o vento vai para a esquerda; então ele vem da direita.
\end{solution}

\begin{solution}{exo:g1:watching-the-weather:11}
Sem folhas: no inverno. Com flores: na primavera. O inverno tem o símbolo
de neve: no inverno faz frio, e em muitos lugares só neva nessa estação.
\end{solution}
