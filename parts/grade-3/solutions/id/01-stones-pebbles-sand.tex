\section*{Bab \ref{ch:g3:stones-pebbles-sand} --- Batu, Kerakal, dan Pasir}

\begin{solution}{exo:g3:stones-pebbles-sand:1}
\omterm{def:g3:stones-pebbles-sand:rock}{Batuan dasar} adalah \omterm{def:g3:stones-pebbles-sand:rock}{batuan} padat yang terletak di bawah tanah di semua
tempat. Kita melihatnya di tempat yang tidak ada tanah yang menutupinya: di
sisi \omterm{def:g1:landscapes-around-us:landform}{gunung} yang curam, di tebing pantai, di sepanjang jalan yang memotong
\omterm{def:g1:landscapes-around-us:landform}{bukit}.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:2}
\omterm{def:g3:stones-pebbles-sand:sand}{Lumpur}, \omterm{def:g3:stones-pebbles-sand:sand}{pasir}, \omterm{def:g3:stones-pebbles-sand:pebble}{kerakal}, bongkah.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:3}
1 milimeter lebih kecil daripada 2 milimeter: \omterm{def:g3:stones-pebbles-sand:sand}{pasir}. 3 milimeter lebih
besar daripada 2 milimeter: \omterm{def:g3:stones-pebbles-sand:pebble}{kerikil}.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:4}
Alami: \omterm{def:g3:stones-pebbles-sand:pebble}{kerakal}, balok granit (dipotong oleh manusia, tetapi \omterm{def:g3:stones-pebbles-sand:rock}{batuannya}
alami), \omterm{def:g3:stones-pebbles-sand:sand}{pasir} di pantai. Buatan manusia: bata dan kaca.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:5}
Air \omterm{def:g1:landscapes-around-us:water}{sungai} sudah lama menggulingkannya di dasar \omterm{def:g1:landscapes-around-us:water}{sungai} dan
membenturkannya ke batu-batu lain: sudut-sudutnya sudah aus.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:6}
Granit, untuk dinding; lempengan batu yang tipis dan datar (\omterm{def:g3:stones-pebbles-sand:rock}{batuan} seperti
batu sabak), untuk atap.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:7}
Yang bulat: makin lama sebuah batu digulingkan air, makin aus
sudut-sudutnya. Yang tajam belum menempuh perjalanan jauh dari \omterm{def:g3:stones-pebbles-sand:rock}{batuan}
asalnya.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:8}
\omterm{def:g3:stones-pebbles-sand:pebble}{Kerakal} yang terbesar panjangnya 3 sentimeter. Satu sentimeter sama dengan
10 milimeter, dan $10 \div 2 = 5$: lima butir berukuran 2 milimeter muat
sepanjang 1 sentimeter.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:9}
Di kolam yang tenang. Air yang deras membuat butiran-butiran \omterm{def:g3:stones-pebbles-sand:sand}{lumpur} yang
sangat kecil terus bergerak; butiran itu baru bisa mengendap di dasar di
tempat airnya hampir diam.
\end{solution}

\begin{solution}{exo:g3:stones-pebbles-sand:10}
Karena semuanya dibuat dari bahan yang diambil dari tanah: bata dari tanah
liat, beton dari \omterm{def:g3:stones-pebbles-sand:sand}{pasir}, \omterm{def:g3:stones-pebbles-sand:pebble}{kerikil}, dan semen, kaca dari \omterm{def:g3:stones-pebbles-sand:sand}{pasir}. Tanah liat,
\omterm{def:g3:stones-pebbles-sand:sand}{pasir}, dan \omterm{def:g3:stones-pebbles-sand:pebble}{kerikil} berasal dari pecahan \omterm{def:g3:stones-pebbles-sand:rock}{batuan}.
\end{solution}
