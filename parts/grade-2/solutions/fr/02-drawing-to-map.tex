\section*{Chapitre \ref{ch:g2:drawing-to-map} --- Du dessin à la carte}

\begin{solution}{exo:g2:drawing-to-map:1}
Tel qu'on le voit d'en haut.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:2}
Les traits bleus sur le mur ouest sont les fenêtres. Dans le coin nord-est,
il y a le placard.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:3}
Un titre, une \omterm{def:g2:drawing-to-map:legend}{légende}, une \omterm{def:g2:drawing-to-map:scale}{échelle} et une \omterm{def:g2:drawing-to-map:directions}{rose des vents} (ou une flèche qui indique le nord).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:4}
Le nord est en haut~; l'ouest est à gauche.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:5}
Un carreau représente 1 mètre~: la table mesure 2 mètres de long et 1 mètre de large.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:6}
Le carré marron est la ferme~; la tache verte est le bois.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:7}
Le bois est au sud de la \omterm{def:g1:landscapes-around-us:water}{rivière}, et à l'est de la ferme.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:8}
Un carreau représente 1 kilomètre~: 10 kilomètres d'ouest en est et
8 kilomètres du sud au nord.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:9}
Trois rangées de trois tables~: 9 tables. La porte est dans la partie sud
du mur est, donc la table la plus proche de la porte est au sud (dans le
coin sud-est de la pièce).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:10}
Le lit mesure 2 carreaux de long. La chambre mesure 4 carreaux sur
3 carreaux~: 3 rangées de 4 carreaux, soit $4 + 4 + 4 = 12$ carreaux.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:11}
«~À gauche~» dépend du côté vers lequel Lena est tournée~: si elle fait
demi-tour, la \omterm{def:g1:landscapes-around-us:water}{rivière} est à sa droite. Pour tout le monde~: «~la \omterm{def:g1:landscapes-around-us:water}{rivière}
est au nord de la ferme~».
\end{solution}
