\section*{第 \ref{ch:g2:water-on-earth} 章——地球上的水}

\begin{solution}{exo:g2:water-on-earth:1}
\omterm{def:g2:water-on-earth:salt-fresh}{咸水}：海、\omterm{def:g2:water-on-earth:ocean}{海洋}。\omterm{def:g2:water-on-earth:salt-fresh}{淡水}：\omterm{def:g1:landscapes-around-us:water}{河流}、雨、
山中的\omterm{def:g1:landscapes-around-us:water}{湖泊}、山上的冰。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:2}
因为地表大部分被\omterm{def:g2:water-on-earth:ocean}{海洋}的蓝色海水覆盖：
从太空看，它是蓝色的。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:3}
$100 - 71 = 29$ 个方格为陆地。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:4}
\omterm{def:g2:water-on-earth:ocean}{海洋}更大；海是较小的一片\omterm{def:g2:water-on-earth:salt-fresh}{咸水}，
常位于\omterm{def:g2:water-on-earth:ocean}{海洋}边缘。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:5}
不能。\omterm{def:g2:water-on-earth:ocean}{海洋}的水是\omterm{def:g2:water-on-earth:salt-fresh}{咸水}，喝了它会使身体
失去水分。水手必须携带\omterm{def:g2:water-on-earth:salt-fresh}{淡水}。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:6}
例如：\omterm{def:g1:landscapes-around-us:water}{河流}中、\omterm{def:g1:landscapes-around-us:water}{湖泊}中、山上和南极洲的冰雪中、
地下、雨中。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:7}
\omterm{def:g2:water-on-earth:salt-fresh}{咸水}：96 桶。\omterm{def:g2:water-on-earth:salt-fresh}{淡水}：$2 + 2 = 4$ 桶。
（严格来说，地下的一部分水也是咸的；图中
把它算作\omterm{def:g2:water-on-earth:salt-fresh}{淡水}。）
\end{solution}

\begin{solution}{exo:g2:water-on-earth:8}
\omterm{def:g2:water-on-earth:ocean}{海洋}（96 桶），然后是冰和雪（2 桶），
最后是\omterm{def:g1:landscapes-around-us:water}{河流}（远少于 1 桶）。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:9}
它可以渗入地下；也可以沿地面流入
溪流或\omterm{def:g1:landscapes-around-us:water}{河流}；太阳出来时，它还可以蒸发回到空气中。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:10}
锅底留下了盐。水已经蒸发了：它以水蒸气的形式
进入了空气中。
\end{solution}

\begin{solution}{exo:g2:water-on-earth:11}
例如：水滴沿屋顶流入檐沟和排水管，
再流入溪流和\omterm{def:g1:landscapes-around-us:water}{河流}；\omterm{def:g1:landscapes-around-us:water}{河流}把它带到海里；某个晴朗
的日子，它从海面蒸发，升入空气中，成为
云的一部分。
\end{solution}
