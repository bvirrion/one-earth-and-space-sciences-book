\section*{Capítulo \ref{ch:g2:drawing-to-map} --- Del dibujo al mapa}

\begin{solution}{exo:g2:drawing-to-map:1}
Tal como se ve desde arriba.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:2}
Las líneas azules de la pared oeste son las ventanas. En la esquina noreste
está el armario.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:3}
Un título, una \omterm{def:g2:drawing-to-map:legend}{leyenda}, una \omterm{def:g2:drawing-to-map:scale}{escala} y una \omterm{def:g2:drawing-to-map:directions}{rosa de los vientos} (o una flecha
que señale el norte).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:4}
El norte está arriba; el oeste, a la izquierda.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:5}
Un cuadrado representa 1 metro: la mesa mide 2 metros de largo y 1 metro de
ancho.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:6}
El cuadrado marrón es la granja; la mancha verde es el bosque.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:7}
El bosque está al sur del \omterm{def:g1:landscapes-around-us:water}{río} y al este de la granja.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:8}
Un cuadrado representa 1 kilómetro: 10 kilómetros de oeste a este y
8 kilómetros de sur a norte.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:9}
Tres filas de tres mesas: 9 mesas. La puerta está en la parte sur de la
pared este, así que la mesa más cercana a la puerta está en el sur (en la
esquina sureste del aula).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:10}
La cama mide 2 cuadrados de largo. La habitación mide 4 cuadrados por
3 cuadrados: 3 filas de 4 cuadrados, es decir, $4 + 4 + 4 = 12$ cuadrados.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:11}
``Izquierda'' depende de hacia dónde mire Lena: si se da la vuelta, el \omterm{def:g1:landscapes-around-us:water}{río}
queda a su derecha. Para todos: ``el \omterm{def:g1:landscapes-around-us:water}{río} está al norte de la granja''.
\end{solution}
