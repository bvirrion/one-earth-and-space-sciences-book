\section*{Chapter \ref{ch:g2:drawing-to-map} --- From a Drawing to a Map}

\begin{solution}{exo:g2:drawing-to-map:1}
As seen from above.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:2}
The blue lines on the west wall are the windows. In the north-east corner
there is the cupboard.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:3}
A title, a \omterm{def:g2:drawing-to-map:legend}{legend}, a \omterm{def:g2:drawing-to-map:scale}{scale} and a \omterm{def:g2:drawing-to-map:directions}{compass rose} (or an arrow pointing north).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:4}
North is at the top; west is on the left.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:5}
One square stands for 1 metre: the desk is 2 metres long and 1 metre wide.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:6}
The brown square is the farm; the green shape is the wood.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:7}
The wood is south of the \omterm{def:g1:landscapes-around-us:water}{river}, and east of the farm.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:8}
One square stands for 1 kilometre: 10 kilometres from west to east and 8
kilometres from south to north.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:9}
Three rows of three desks: 9 desks. The door is in the south part of the
east wall, so the desk nearest to the door is in the south (the south-east
corner of the room).
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:10}
The bed is 2 squares long. The room is 4 squares by 3 squares: 3 rows of 4
squares, that is $4 + 4 + 4 = 12$ squares.
\end{solution}

\begin{solution}{exo:g2:drawing-to-map:11}
``Left'' depends on which way Lena is facing: if she turns round, the \omterm{def:g1:landscapes-around-us:water}{river}
is on her right. For everybody: ``the \omterm{def:g1:landscapes-around-us:water}{river} is north of the farm''.
\end{solution}
